def parse_23andme(file_content):
    """
    23andMe raw data에서 주요 SNP 30개 추출
    """
    target_snps = {
        'rs4680': 'COMT',      # Val158Met
        'rs1801133': 'MTHFR',  # C677T
        'rs4880': 'SOD2',      # Ala16Val
        'rs1799752': 'ACE',    # I/D
        'rs6265': 'BDNF',      # Val66Met
        # ... 25개 더
    }

    results = {}

    for line in file_content:
        if line.startswith('#'):
            continue

        rsid, chrom, pos, genotype = line.split('\t')

        if rsid in target_snps:
            results[rsid] = {
                'gene': target_snps[rsid],
                'genotype': genotype,
                'chromosome': chrom,
                'position': pos
            }

    return results

class BaselineCalculator:
    """
    유전자 데이터 → 개인 기준선 계산
    """
    def calculate_baselines(self, genetic_variants):
        baselines = {}

        # HRV 기준선
        baselines['hrv'] = self._calculate_hrv_baseline(
            comt=genetic_variants['COMT'],
            bdnf=genetic_variants['BDNF'],
            ace=genetic_variants['ACE']
        )

        # 회복 시간 기준선
        baselines['recovery'] = self._calculate_recovery_baseline(
            sod2=genetic_variants['SOD2'],
            ppargc1a=genetic_variants['PPARGC1A']
        )

        # 운동 타입 추천
        baselines['exercise'] = self._determine_exercise_type(
            ace=genetic_variants['ACE'],
            actn3=genetic_variants['ACTN3']
        )

        # 영양소 필요량
        baselines['nutrients'] = self._calculate_nutrient_needs(
            mthfr=genetic_variants['MTHFR'],
            vdr=genetic_variants['VDR'],
            comt=genetic_variants['COMT']
        )

        return baselines

    def _calculate_hrv_baseline(self, comt, bdnf, ace):
        base = 50  # 평균 성인

        # COMT 영향
        if comt == 'AA':  # Met/Met
            base -= 10
        elif comt == 'GG':  # Val/Val
            base += 10

        # BDNF 영향
        if bdnf in ['AG', 'AA']:  # Met carrier
            base -= 5

        # ACE 영향
        if ace == 'II':
            base += 5
        elif ace == 'DD':
            base -= 3

        return {
            'optimal': base,
            'range': (base - 10, base + 10),
            'stress_threshold': base - 8
        }

```python
class RealtimeAnalyzer:
    """
    실시간 Apple Watch 데이터 + 유전자 기준선 → 해석
    """
    def analyze_hrv(self, user_id, current_hrv):
        # 1. 유저 데이터 로드
        genetics = db.collection('users').document(user_id)\
                     .collection('genetics').get()
        baseline = db.collection('users').document(user_id)\
                     .collection('baselines').get()

        # 2. 유전자 기반 해석
        comt_genotype = genetics['COMT']['genotype']
        expected_hrv = baseline['hrv']['optimal']

        deviation = current_hrv - expected_hrv

        # 3. 컨텍스트 분석
        recent_activities = self._get_recent_activities(user_id, hours=24)
        sleep_last_night = self._get_sleep_data(user_id, last_night=True)

        # 4. 해석 생성
        if abs(deviation) <= 5:
            status = "NORMAL"
            message = f"HRV {current_hrv}는 당신의 기준 ({expected_hrv})에 적합합니다."

        elif deviation < -10:
            status = "LOW"

            # 원인 분석
            causes = []
            if comt_genotype == 'AA' and recent_activities['stress_events'] > 0:
                causes.append(
                    "COMT Met/Met 유전자는 스트레스에 민감합니다. "
                    "오늘 스트레스 이벤트가 감지되었습니다."
                )

            if sleep_last_night['duration'] < 7:
                causes.append(
                    f"수면 부족 ({sleep_last_night['duration']}시간)"
                )

            if recent_activities['intense_workout_hours_ago'] < 24:
                causes.append("어제 고강도 운동의 영향")

            message = " + ".join(causes)

        # 5. 추천 생성
        recommendations = self._generate_recommendations(
            status, comt_genotype, current_hrv, expected_hrv
        )

        return {
            'status': status,
            'current': current_hrv,
            'expected': expected_hrv,
            'deviation': deviation,
            'message': message,
            'recommendations': recommendations,
            'genetic_context': self._explain_genetic_influence(comt_genotype)
        }
```

class PersonalizedPatternModel:
    """
    사용자별 데이터 축적 → 패턴 학습 → 예측
    """
    def train_user_model(self, user_id):
        # 1. 사용자 데이터 수집 (최소 30일)
        history = db.collection('users').document(user_id)\
                    .collection('health_data')\
                    .where('timestamp', '>=', datetime.now() - timedelta(days=90))\
                    .get()

        # 2. 피처 엔지니어링
        features = []
        labels = []

        for day in history:
            # 입력 피처
            X = {
                'hrv': day['hrv_avg'],
                'sleep_hours': day['sleep_duration'],
                'workout_intensity': day['workout_intensity'],
                'stress_events': day['stress_count'],
                'supplements_taken': day['supplements'],

                # 유전자 정보 (고정값)
                'comt': genetics['COMT_encoded'],
                'sod2': genetics['SOD2_encoded'],
                'mthfr': genetics['MTHFR_encoded']
            }

            # 레이블 (다음날 상태)
            y = {
                'next_day_hrv': next_day['hrv_avg'],
                'next_day_energy': next_day['energy_level'],
                'recovery_time': next_day['recovery_percentage']
            }

            features.append(X)
            labels.append(y)

        # 3. 모델 학습 (XGBoost)
        model = xgb.XGBRegressor()
        model.fit(features, labels)

        # 4. 모델 저장
        model_path = f'gs://metabo-lens/models/user_{user_id}.pkl'
        joblib.dump(model, model_path)

        return model

    def predict_tomorrow(self, user_id, today_data):
        """
        오늘 데이터 → 내일 상태 예측
        """
        model = self._load_user_model(user_id)

        prediction = model.predict([today_data])

        return {
            'predicted_hrv': prediction['next_day_hrv'],
            'predicted_energy': prediction['next_day_energy'],
            'confidence': model.predict_proba([today_data]),

            'advice': (
                f"오늘 수면을 {today_data['sleep_hours']}시간 자면 "
                f"내일 HRV는 {prediction['next_day_hrv']:.0f} 예상됩니다. "
                f"8시간 자면 +5 증가 예상."
            )
        }

# 사용 예시
prediction = predictor.predict_tomorrow(
    user_id='user123',
    today_data={
        'hrv': 42,
        'sleep_hours': 7.5,
        'workout_intensity': 6,
        'stress_events': 1,
        'supplements_taken': ['nmn', 'methylfolate', 'magnesium']
    }
)
